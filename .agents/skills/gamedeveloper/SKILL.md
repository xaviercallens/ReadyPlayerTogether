## 1. Ready Player Me Avatars & Viseme Lip-Sync

OASIS NPCs (Parzival, Art3mis, Aech) and player avatars utilize Ready Player Me (RPM) GLTF models with facial blend shape visemes.

### Viseme Lip-Sync GDScript Component
```gdscript
class_name AvatarLipSync
extends Node

@export var mesh_instance: MeshInstance3D
@export var audio_player: AudioStreamPlayer3D

# Viseme Blendshape Mapping
var viseme_names = [
	"viseme_sil", "viseme_PP", "viseme_FF", "viseme_TH",
	"viseme_DD", "viseme_kk", "viseme_CH", "viseme_SS",
	"viseme_nn", "viseme_RR", "viseme_aa", "viseme_E",
	"viseme_I", "viseme_O", "viseme_U"
]

func speak_viseme(viseme_name: String, weight: float) -> void:
	if not mesh_instance:
		return
	var mesh = mesh_instance.mesh
	if not mesh:
		return
		
	var idx = mesh_instance.find_blend_shape_by_name(viseme_name)
	if idx != -1:
		mesh_instance.set_blend_shape_value(idx, weight)

func reset_visemes() -> void:
	if not mesh_instance:
		return
	for v_name in viseme_names:
		var idx = mesh_instance.find_blend_shape_by_name(v_name)
		if idx != -1:
			mesh_instance.set_blend_shape_value(idx, 0.0)
```

## 2. Vehicle Mechanics (DeLorean Controller)

The DeLorean vehicle controller combines 3D RigidBody physics with hover mechanics and hologram spawner visual effects.

```gdscript
class_name DeLoreanVehicle
extends RigidBody3D

@export var engine_power: float = 1200.0
@export var turn_speed: float = 2.5
@export var hover_height: float = 0.5
@export var hover_force: float = 150.0

@onready var raycast_front: RayCast3D = $RayCastFront
@onready var raycast_back: RayCast3D = $RayCastBack

func _physics_process(delta: float) -> void:
	# Hover Raycasts
	_apply_hover(raycast_front)
	_apply_hover(raycast_back)
	
	# Driving Input
	var throttle = Input.get_axis("move_backward", "move_forward")
	var steer = Input.get_axis("move_right", "move_left")
	
	if throttle != 0:
		apply_central_force(-transform.basis.z * throttle * engine_power)
	if steer != 0:
		apply_torque(Vector3.UP * steer * turn_speed * 100.0)

func _apply_hover(ray: RayCast3D) -> void:
	if ray and ray.is_colliding():
		var hit_point = ray.get_collision_point()
		var distance = ray.global_position.distance_to(hit_point)
		if distance < hover_height:
			var force_magnitude = (1.0 - (distance / hover_height)) * hover_force
			apply_force(Vector3.UP * force_magnitude, ray.global_position - global_position)
```

## 3. World Generation Import (`AgibotTech/genie_sim`)

When incorporating 3D environments generated by `AgibotTech/genie_sim`:
1. Ensure the exported `.gltf` or `.usd` file is placed in `assets/models/genie_sim/`.
2. Python script `import_rpo_godot_assets.py` registers the model resource in Godot.
3. Automatically generate collision meshes for static geometry:
```gdscript
func setup_genie_sim_world(world_mesh_instance: MeshInstance3D) -> void:
	if world_mesh_instance and world_mesh_instance.mesh:
		world_mesh_instance.create_trimesh_collision()
		print("[OASIS GameDev] Created collision mesh for GenieSim world.")
```

## 5. Quest 3S Performance & Shader Rules
- Target Framerate: **60 FPS** (90 FPS on VR link).
- Keep total draw calls under **150 per scene**.
- Avoid real-time volumetric fog on mobile/Quest 3S build targets; use distance-fade shaders instead.
- Enable VRAM texture compression ASTC in `project.godot` (`textures/vram_compression/import_etc2_astc=true`).
